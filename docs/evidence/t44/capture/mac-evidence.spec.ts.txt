// Evidence capture for T44 in the Mac app under xvfb (kept as
// docs/evidence/t44/capture/mac-evidence.spec.ts.txt; not a suite spec). OUT=dir. The window
// keeps Linux's frame here (A725); the page is the Mac app's.
import { expect, test, type ElectronApplication, type Page } from '@playwright/test';
import { mkdirSync } from 'node:fs';
import { body, launch, mainBounds, removeDataDirs, signIn, ULID } from './support';

const OUT = process.env['OUT'] ?? '/tmp/t44-evidence';
mkdirSync(OUT, { recursive: true });
const RUN = Date.now().toString(36).slice(-4);
const mac = process.platform === 'darwin';
const NEW_NOTE = mac ? 'Meta+Alt+KeyN' : 'Alt+KeyN';
const SYMBOL = (slot: number) => (mac ? `Control+Digit${slot}` : `Alt+Shift+Digit${slot}`);

test.afterAll(() => removeDataDirs());

const chooseMenu = (app: ElectronApplication, id: string) =>
  app.evaluate(({ Menu }, itemId) => {
    (Menu.getApplicationMenu()?.getMenuItemById(itemId) as unknown as { click(): void }).click();
  }, id);

async function scheme(app: ElectronApplication, page: Page, value: 'light' | 'dark') {
  await app.evaluate(({ nativeTheme }, s) => {
    nativeTheme.themeSource = s;
  }, value);
  await page.emulateMedia({ colorScheme: value });
  await expect(page.locator('html')).toHaveAttribute('data-mode', value);
  await page.mouse.move(1270, 790);
}

test('T44 mac evidence', async () => {
  test.setTimeout(300_000);
  const launched = await launch();
  const { app, page } = launched;
  try {
    await mainBounds(app, { x: 0, y: 0, width: 1280, height: 800 });
    await signIn(launched);
    await expect(page.locator('button[data-command="note.new"]').first()).toBeVisible();
    await page.keyboard.press(NEW_NOTE);
    await expect(page).toHaveURL(ULID);
    await page.keyboard.type(`Week plan ${RUN}`);
    await page.keyboard.press('Enter');
    await page.keyboard.type('Call the bank about the mortgage before Friday');
    await chooseMenu(app, 'editor.symbol1');
    await page.keyboard.press('Enter');
    await page.keyboard.type('→ Book the plumber for the kitchen tap');
    await page.keyboard.press('Enter');
    await page.keyboard.type('## ? Open questions');
    await page.keyboard.press('Enter');
    await page.keyboard.type('- Which week suits the builders?');
    await page.keyboard.press('Enter');
    await page.keyboard.type('Do we need a permit for the shed?');
    await page.keyboard.press(SYMBOL(2));
    await page.keyboard.press('Enter');
    await page.keyboard.press('Enter');
    await page.keyboard.type('[ ] Send the form to the council');
    await page.keyboard.press(SYMBOL(3));
    await page.keyboard.press(SYMBOL(1));
    await page.keyboard.press('Enter');
    await page.keyboard.type('Water the beds');
    await page.keyboard.press('Enter');
    await page.keyboard.press('Enter');
    await page.keyboard.type('💡 A rain barrel by the shed would save the hose.');
    await expect(body(page).locator('.pm-symbol')).toHaveCount(7);
    for (const value of ['light', 'dark'] as const) {
      await scheme(app, page, value);
      await page.waitForTimeout(400);
      await page.screenshot({ path: `${OUT}/mac-${value}-gutter.png` });
      await body(page).locator('li', { hasText: 'Send the form' }).locator('p').first().click();
      // As a person reaches for the menu: the editor has read the click's caret by then.
      await page.waitForTimeout(300);
      await chooseMenu(app, 'editor.symbolPicker');
      await expect(page.getByTestId('symbol-picker')).toBeVisible();
      await page.waitForTimeout(300);
      await page.screenshot({ path: `${OUT}/mac-${value}-picker.png` });
      await page.keyboard.press('Escape');
    }
    expect(launched.errors).toEqual([]);
  } finally {
    await app.close();
  }
});
